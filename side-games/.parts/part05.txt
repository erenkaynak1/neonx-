async function createLobby(name){
  let code;
  for(let tries=0; tries<5; tries++){
    code = makeCode();
    try{
      const existing = await window.storage.get('lobby:'+code, true);
      if(!existing) break;
    }catch(e){ break; }
  }
  const O = state.online;
  O.isHost = true;
  O.code = code;
  O.myName = name;
  O.players = [{ id: O.myId, name }];
  O.status = 'waiting';
  O.round = 0;
  try{
    await window.storage.set('lobby:'+code, JSON.stringify({
      code, hostId: O.myId, status: 'waiting', players: O.players,
      round: 0, footballer: null, imposterId: null
    }), true);
    O.err='';
    state.screen = 'online-lobby';
    startPolling();
    render();
  }catch(e){
    O.err = 'Lobi oluşturulamadı, tekrar dene.';
    state.screen = 'online-create';
    render();
  }
}

  };

async function joinLobby(code, name){
  const O = state.online;
  try{
    const res = await window.storage.get('lobby:'+code, true);
    if(!res){ O.err='Böyle bir lobi bulunamadı.'; state.screen='online-join'; render(); return; }
    const data = JSON.parse(res.value);
    if(data.status === 'playing'){ O.err='Bu lobide oyun zaten başladı.'; state.screen='online-join'; render(); return; }
    data.players.push({ id: O.myId, name });
    await window.storage.set('lobby:'+code, JSON.stringify(data), true);
    O.isHost = false;
    O.code = code;
    O.myName = name;
    O.players = data.players;
    O.status = data.status;
    O.err = '';
    state.screen = 'online-lobby';
    startPolling();
    render();
  }catch(e){
    O.err = 'Bağlantı hatası, tekrar dene.';
    state.screen = 'online-join';
    render();
  }
}
