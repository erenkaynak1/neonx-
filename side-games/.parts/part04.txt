ibility*0.28).toFixed(3) + ')' +
      ' saturate(' + (0.92 + visibility*0.12).toFixed(3) + ')' +
      ' drop-shadow(' + (-foldNX*7*visibility).toFixed(1) + 'px ' + (-foldNY*7*visibility).toFixed(1) + 'px ' + (5+visibility*7).toFixed(1) + 'px rgba(0,0,0,' + (0.12+visibility*0.26).toFixed(3) + '))';

    // Ön yüzde çok hafif ışık değişimi; düz dijital maske hissini azaltır.
    front.style.filter = 'brightness(' + (1 - visibility*0.035).toFixed(3) + ')';
  }

  function progressFromPointer(e){
    const rect = wrap.getBoundingClientRect();
    const innerW = Math.max(1, rect.width - 20);
    const innerH = Math.max(1, rect.height - 20);
    const dx = Math.max(0, startX - e.clientX);
    const dy = Math.max(0, startY - e.clientY);
    const px = dx / innerW;
    const py = dy / innerH;
    const p = clamp((px + py) / 2);

    // Kullanıcı daha yatay/dikey çekerse fold açısı da buna doğal şekilde tepki verir.
    // Çok küçük harekette açı zıplamasın diye mevcut normal korunur.
    if(dx + dy > 4){
      const vx = Math.max(0.08, px);
      const vy = Math.max(0.08, py);
      const len = Math.hypot(vx,vy) || 1;
      foldNX = vx/len;
      foldNY = vy/len;
    }
    return p;
  }

  function stopSnap(){
    if(snapRaf){ cancelAnimationFrame(snapRaf); snapRaf = null; }
  }

  function snapTo(target){
    stopSnap();
    const from = progress;
    const duration = 220;
    const t0 = performance.now();
    function frame(now){
      const t = clamp((now - t0) / duration);
      const eased = 1 - Math.pow(1-t,3);
      renderProgress(from + (target-from)*eased);
      if(t < 1){
        snapRaf = requestAnimationFrame(frame);
      } else {
        snapRaf = null;
        progress = target;
        opened = target === 1;
        hitZone.classList.toggle('disabled', opened);
        onChange(opened);
      }
    }
    snapRaf = requestAnimationFrame(frame);
  }

  function onDown(e){
    if(opened) return;
    stopSnap();
    dragging = true;
    pointerId = e.pointerId;
    startX = e.clientX;
    startY = e.clientY;
    try{ hitZone.setPointerCapture(pointerId); }catch(err){}
    e.preventDefault();
  }

  function onMove(e){
    if(!dragging || e.pointerId !== pointerId) return;
    // Drag sırasında hiçbir transition/snap yok: geometri doğrudan parmağa bağlı.
    renderProgress(progressFromPointer(e), foldNX, foldNY);
    e.preventDefault();
  }

  function finishDrag(e){
    if(!dragging || (pointerId !== null && e.pointerId !== pointerId)) return;
    dragging = false;
    try{ hitZone.releasePointerCapture(pointerId); }catch(err){}
    pointerId = null;
    const target = progress > 0.55 ? 1 : 0;
    snapTo(target);
    e.preventDefault();
  }

  hitZone.addEventListener('pointerdown', onDown);
  hitZone.addEventListener('pointermove', onMove);
  hitZone.addEventListener('pointerup', finishDrag);
  hitZone.addEventListener('pointercancel', finishDrag);

  // İlk render: revealed state korunur.
  requestAnimationFrame(() => renderProgress(progress));

  return wrap;
}

function syncLocalNames(){
  const n = state.local.count;
  while(state.local.names.length < n) state.local.names.push('');
  while(state.local.names.length > n) state.local.names.pop();
}

function startLocalGame(){
  const names = state.local.names.map((n,i) => n.trim() || ('Oyuncu ' + (i+1)));
  state.local.footballer = pick(FOOTBALLERS);
  state.local.imposterIdx = Math.floor(Math.random()*names.length);
  state.local.order = names;
  state.local.idx = 0;
  state.local.revealedThis = false;
  state.screen = 'local-reveal';
  render();
}
