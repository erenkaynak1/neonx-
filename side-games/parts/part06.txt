/* ---------- ONLINE: START ROUND ---------- */
function startOnlineRound(){
  const O = state.online;
  if(!O.isHost || O.players.length < 3) return;
  const footballer = pick(FOOTBALLERS);
  const imposter = pick(O.players).id;
  const newRound = O.round + 1;

  O.status = 'playing';
  O.footballer = footballer;
  O.imposterId = imposter;
  O.round = newRound;
  O.revealed = false;
  broadcastLobbyState();
  state.screen = 'online-reveal';
  render();
}

/* ---------- ONLINE: REVEAL ---------- */
function renderOnlineReveal(app){
  const O = state.online;
  const isImposter = O.imposterId === O.myId;
  header(app, 'TUR ' + O.round, O.myName.toUpperCase(), 'Rolünü kimse görmesin');

  const hideBtn = el('button','btn ghost','GİZLE');
  const peel = buildPeelCard(O.revealed, isImposter, O.footballer, (revealed) => {
    O.revealed = revealed;
    hideBtn.classList.toggle('hidden', !revealed);
  });
  app.appendChild(peel);
  app.appendChild(el('div','peel-caption','Kartın sağ alt köşesini yukarı doğru kaydır'));

  hideBtn.classList.toggle('hidden', !O.revealed);
  hideBtn.onclick = () => { O.revealed = false; render(); };
  app.appendChild(hideBtn);

  if(O.isHost){
    const again = el('button','btn solid','YENİ TUR BAŞLAT');
    again.onclick = () => startOnlineRound();
    app.appendChild(again);
  } else {
    app.appendChild(el('div','info-text','Lobi sahibi yeni tur başlattığında ekranın otomatik güncellenecek.'));
  }

  const leave = el('button','btn danger','LOBİDEN AYRIL');
  leave.onclick = () => leaveLobby();
  app.appendChild(leave);
}

/* ---------- ONLINE: LEAVE ---------- */
function leaveLobby(){
  const O = state.online;
  p2pLeaving = true;
  try{
    if(O.isHost){
      p2pHostConnections.forEach(conn => sendConn(conn,{type:'host-closed'}));
    } else if(p2pHostConnection){
      sendConn(p2pHostConnection,{type:'leave', playerId:O.myId});
    }
  }catch(e){}

  setTimeout(() => {
    clearP2P();
    p2pLeaving = false;
  },60);

  O.code='';
  O.players=[];
  O.status='idle';
  O.isHost=false;
  O.hostId=null;
  O.footballer=null;
  O.imposterId=null;
  O.round=0;
  O.revealed=false;
  O.err='';
  state.screen = 'menu';
  render();
}

window.addEventListener('beforeunload', () => {
  p2pLeaving = true;
  try{
    if(state.online.isHost){
      p2pHostConnections.forEach(conn => sendConn(conn,{type:'host-closed'}));
    } else if(p2pHostConnection){
      sendConn(p2pHostConnection,{type:'leave', playerId:state.online.myId});
    }
  }catch(e){}
  clearP2P();
});

render();
