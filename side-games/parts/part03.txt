  function clipHalfPlane(poly, nx, ny, threshold, keepLess){
    if(!poly.length) return [];
    const out = [];
    const side = (pt) => nx*pt[0] + ny*pt[1] - threshold;
    const inside = (pt) => keepLess ? side(pt) <= 1e-7 : side(pt) >= -1e-7;
    for(let i=0;i<poly.length;i++){
      const a = poly[i];
      const b = poly[(i+1)%poly.length];
      const aIn = inside(a);
      const bIn = inside(b);
      if(aIn) out.push(a);
      if(aIn !== bIn){
        const sa = side(a);
        const sb = side(b);
        const denom = sa - sb;
        const t = Math.abs(denom) < 1e-9 ? 0 : sa / denom;
        out.push([
          a[0] + (b[0]-a[0]) * t,
          a[1] + (b[1]-a[1]) * t
        ]);
      }
    }
    return out;
  }

  function polyCss(poly){
    if(poly.length < 3) return 'polygon(0 0,0 0,0 0)';
    return 'polygon(' + poly.map(([x,y]) => (x*100).toFixed(3) + '% ' + (y*100).toFixed(3) + '%').join(',') + ')';
  }

  // nx*x + ny*y = threshold çizgisinin kart kenarlarıyla kesişimlerini bulur.
  function lineSquareIntersections(nx, ny, threshold){
    const pts = [];
    const add = (x,y) => {
      if(x < -1e-6 || x > 1+1e-6 || y < -1e-6 || y > 1+1e-6) return;
      if(!pts.some(p => Math.hypot(p[0]-x,p[1]-y) < 1e-5)) pts.push([clamp(x),clamp(y)]);
    };
    if(Math.abs(ny) > 1e-8){
      add(0, threshold/ny);
      add(1, (threshold-nx)/ny);
    }
    if(Math.abs(nx) > 1e-8){
      add(threshold/nx, 0);
      add((threshold-ny)/nx, 1);
    }
    if(pts.length < 2) return [[1,1],[1,1]];
    let best=[pts[0],pts[1]], bestD=-1;
    for(let i=0;i<pts.length;i++) for(let j=i+1;j<pts.length;j++){
      const d=Math.hypot(pts[i][0]-pts[j][0],pts[i][1]-pts[j][1]);
      if(d>bestD){ bestD=d; best=[pts[i],pts[j]]; }
    }
    return best;
  }
